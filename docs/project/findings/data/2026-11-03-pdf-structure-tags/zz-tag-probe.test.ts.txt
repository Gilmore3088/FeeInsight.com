// @vitest-environment node
// Measurement probe for findings/2026-11-03-pdf-structure-tags.md: a tag-based PDF reading of each
// page whose priced text sits wholly in Table cells, else main's layout reading, run through the
// source-bytes replay. To run: copy into src/lib/agents/deming/shadow-lab/ as a .test.ts file.
import { readFileSync, writeFileSync, mkdirSync } from "node:fs";
import { join } from "node:path";
import { describe, expect, it, vi } from "vitest";

vi.mock("@/lib/ai-provider", async (importOriginal) => {
  const real = await importOriginal<Record<string, unknown>>();
  return Object.fromEntries(Object.entries(real).map(([n, v]) => [n, typeof v === "function" ? () => { throw new Error(`refused ${n}`); } : v]));
});

const OUT = process.env.TAG_PROBE_OUT ?? "test-results/tag-probe";
const PRICE = /\$\s?\d|\d+\.\d{2}\b/;
const CELL_ROLES = new Set(["TD", "TH"]);
const HEADING = /^H[1-6]?$/;

interface Node { role?: string; type?: string; id?: string; children?: Node[] }

async function tagPages(bytes: Uint8Array, mode: "tables" | "all") {
  const { getDocumentProxy } = await import("unpdf");
  const pdf = await getDocumentProxy(new Uint8Array(bytes));
  const out: { page: number; used: boolean; why: string; text: string | null }[] = [];
  for (let p = 1; p <= pdf.numPages; p += 1) {
    const page = await pdf.getPage(p);
    const tree = (await page.getStructTree().catch(() => null)) as Node | null;
    const content = await page.getTextContent({ includeMarkedContent: true });
    const idText = new Map<string, string[]>();
    const stack: (string | null)[] = [];
    const priced: { str: string; ids: string[] }[] = [];
    for (const item of content.items as Array<Record<string, unknown>>) {
      if (item.type === "beginMarkedContentProps" || item.type === "beginMarkedContent") stack.push((item.id as string) ?? null);
      else if (item.type === "endMarkedContent") stack.pop();
      else if (typeof item.str === "string") {
        const ids = stack.filter((v): v is string => Boolean(v));
        const id = ids[ids.length - 1];
        if (id) { const list = idText.get(id) ?? []; list.push(item.str + (item.hasEOL ? " " : "")); idText.set(id, list); }
        if (PRICE.test(item.str)) priced.push({ str: item.str, ids });
      }
    }
    if (!tree?.children?.length) { out.push({ page: p, used: false, why: "untagged", text: null }); continue; }
    const cellIds = new Set<string>();
    const anyIds = new Set<string>();
    const collect = (n: Node, inCell: boolean) => {
      const cell = inCell || CELL_ROLES.has(n.role ?? "") || (mode === "all" && (n.role === "LI" || HEADING.test(n.role ?? "")));
      if (n.type === "content" && n.id) { anyIds.add(n.id); if (cell) cellIds.add(n.id); }
      for (const c of n.children ?? []) collect(c, cell);
    };
    collect(tree, false);
    const outside = priced.filter((it) => !it.ids.some((id) => cellIds.has(id)));
    if (priced.length === 0) { out.push({ page: p, used: false, why: "no priced text", text: null }); continue; }
    if (outside.length > 0) { out.push({ page: p, used: false, why: `priced outside cell ${outside.length}/${priced.length}`, text: null }); continue; }
    const textOf = (n: Node): string => {
      if (n.type === "content" && n.id) return (idText.get(n.id) ?? []).join("");
      return (n.children ?? []).map(textOf).filter((s) => s.trim()).join(" ");
    };
    const lines: string[] = [];
    const emit = (n: Node) => {
      const role = n.role ?? "";
      if (role === "TR") { lines.push((n.children ?? []).map((c) => textOf(c).replace(/\s+/g, " ").trim()).filter(Boolean).join(" | ")); return; }
      if (HEADING.test(role) || role === "P" || role === "LI" || role === "Caption") { lines.push(textOf(n).replace(/\s+/g, " ").trim()); return; }
      if (n.type === "content") { lines.push(textOf(n).replace(/\s+/g, " ").trim()); return; }
      for (const c of n.children ?? []) emit(c);
    };
    emit(tree);
    out.push({ page: p, used: true, why: "covered", text: lines.filter(Boolean).join("\n") });
  }
  await pdf.destroy?.();
  return out;
}

describe("tag probe", () => {
  it("measures", async () => {
    mkdirSync(OUT, { recursive: true });
    const { applyManifest, loadScenarios } = await import("./scenarios");
    const { currentVariant } = await import("./variants");
    const { readSourceManifest, rereadAcceptedSources, loadSourceBytes } = await import("./source-bytes");
    const { replaySources, replayDelta } = await import("./source-replay");
    const { extractPdfDocumentText } = await import("@/lib/agents/rosetta/pdf-document-text");
    const { normalizeWhitespace } = await import("@/lib/agents/rosetta/read");
    const frozen = JSON.parse(readFileSync(join(__dirname, "frozen-inputs.json"), "utf8")) as Record<string, string>;
    const cases = applyManifest(loadScenarios(), frozen);
    const manifest = readSourceManifest();
    const base = await rereadAcceptedSources();
    const report: string[] = [];
    for (const mode of ["tables"] as const) {
      const on = new Map(base);
      for (const entry of manifest.accepted) {
        const read = base.get(entry.caseKey);
        if (!read?.ok || read.reader !== "read.pdf_layout") continue;
        const bytes = loadSourceBytes(entry);
        const main = await extractPdfDocumentText(bytes);
        const tags = await tagPages(bytes, mode);
        const pages = (main.pages ?? []).map((t, i) => tags[i]?.used ? tags[i].text ?? t : t);
        const used = tags.filter((t) => t.used).length;
        report.push(`${mode} ${entry.caseKey}: tag pages ${used}/${tags.length} ${tags.map((t) => t.why).join("; ")}`);
        if (used === 0) continue;
        const text = normalizeWhitespace(pages.join("\n\n"));
        writeFileSync(join(OUT, `${mode}-${entry.caseKey.replace(/[:]/g, "_")}.tag.txt`), text);
        writeFileSync(join(OUT, `${mode}-${entry.caseKey.replace(/[:]/g, "_")}.main.txt`), read.text);
        on.set(entry.caseKey, { ...read, text });
      }
      const a = replaySources({ manifest, cases, rereads: base, variant: currentVariant });
      const b = replaySources({ manifest, cases, rereads: on, variant: currentVariant });
      const d = replayDelta(a, b);
      report.push(`${mode} recall off ${a.total.reread.found}/${a.total.reread.expected} on ${b.total.reread.found}/${b.total.reread.expected}; outside ${a.total.reread.unexpected} -> ${b.total.reread.unexpected}`);
      report.push(`${mode} DELTA ${JSON.stringify(d)}`);
    }
    writeFileSync(join(OUT, "report.txt"), report.join("\n"));
    expect(true).toBe(true);
  }, 600_000);
});
